# renderer/Direct3D.cpp d3d_GetAspectRatio
float d3d_GetAspectRatio ()
{
	float aspect_ratio = (float)((3.0 * D3D_state.screen_width)/(4.0 * D3D_state.screen_height));
	return aspect_ratio;
}
# renderer/HardwareOpenGL.cpp rend_GetAspectRatio
// Returns the aspect ratio of the physical screen
float rend_GetAspectRatio(void)
{
	float aspect_ratio = (float)((3.0f * OpenGL_state.screen_width)/(4.0f * OpenGL_state.screen_height));
	return aspect_ratio;
}
